35void ptp_nsd_igmp_join_leave(bool join) {
36 // only join IGMP if Transport Type is IP
37 if (TP == PTP_TP_IPv4) {
38 err_t (*igmp_fn)(const ip_addr_t *, const ip_addr_t *) = join ? igmp_joingroup : igmp_leavegroup; // join or leave
39
40 igmp_fn(&netif_default->ip_addr, &PTP_IGMP_PRIMARY); // join E2E DM message group
41
42 if (DM == PTP_DM_P2P) {
43 igmp_fn(&netif_default->ip_addr, &PTP_IGMP_PEER_DELAY); // join P2P DM message group
44 }
45 }

48void ptp_nsd_init(const NsdInitSettings * init) {
49 // lock LWIP core
50 LOCK_TCPIP_CORE();
51
52 // leave current IGMP group if applicable
53 ptp_nsd_igmp_join_leave(false);
54
55 // first, close all open connection blocks (zero CBDs won't cause trouble)
56 if (PTP_L4_PRIMARY_EVENT != NULL) {
57 udp_disconnect(PTP_L4_PRIMARY_EVENT);
58 udp_remove(PTP_L4_PRIMARY_EVENT);
59 PTP_L4_PRIMARY_EVENT = NULL;
60 }
61 if (PTP_L4_PRIMARY_GENERAL != NULL) {
62 udp_disconnect(PTP_L4_PRIMARY_GENERAL);
63 udp_remove(PTP_L4_PRIMARY_GENERAL);
64 PTP_L4_PRIMARY_GENERAL = NULL;
65 }
66 if (PTP_L4_PDELAY_EVENT != NULL) {
67 udp_disconnect(PTP_L4_PDELAY_EVENT);
68 udp_remove(PTP_L4_PDELAY_EVENT);
69 PTP_L4_PDELAY_EVENT = NULL;
70 }
71 if (PTP_L4_PDELAY_GENERAL != NULL) {
72 udp_disconnect(PTP_L4_PDELAY_GENERAL);
73 udp_remove(PTP_L4_PDELAY_GENERAL);
74 PTP_L4_PDELAY_GENERAL = NULL;
75 }
76
77 // calling either parameter with -1 just closes connections
78 if ((init->tp == -1) || (init->dm == -1)) {
79 // message transmission and reception is turned off
80 TP = -1;
81 DM = -1;
82 return;
83 }
84
85 // open only the necessary ones
86 if (init->tp == PTP_TP_IPv4) {
87 // open event and general PRIMARY connections
88 PTP_L4_PRIMARY_EVENT = udp_new();
89 udp_bind(PTP_L4_PRIMARY_EVENT, &PTP_IGMP_PRIMARY, PTP_PORT_EVENT);
90 udp_recv(PTP_L4_PRIMARY_EVENT, ptp_receive_cb, NULL);
91
92 PTP_L4_PRIMARY_GENERAL = udp_new();
93 udp_bind(PTP_L4_PRIMARY_GENERAL, &PTP_IGMP_PRIMARY, PTP_PORT_GENERAL);
94 udp_recv(PTP_L4_PRIMARY_GENERAL, ptp_receive_cb, NULL);
95
96 // open event and general PDELAY* connections
97 if (init->dm == PTP_DM_P2P) {
98 PTP_L4_PDELAY_EVENT = udp_new();
99 udp_bind(PTP_L4_PDELAY_EVENT, &PTP_IGMP_PEER_DELAY, PTP_PORT_EVENT);
100 udp_recv(PTP_L4_PDELAY_EVENT, ptp_receive_cb, NULL);
101
102 PTP_L4_PDELAY_GENERAL = udp_new();
103 udp_bind(PTP_L4_PDELAY_GENERAL, &PTP_IGMP_PEER_DELAY, PTP_PORT_GENERAL);
104 udp_recv(PTP_L4_PDELAY_GENERAL, ptp_receive_cb, NULL);
105 }
106 }
107
108 // if custom P2P 802.3 destination are given, store them
109 uint8_t mac_size = sizeof(zero_mac);
110 if (memcmp(init->primary_p2p_8023_dest, zero_mac, mac_size)) {
111 memcpy(custom_p2p_8023_primary_dest, &init->primary_p2p_8023_dest, mac_size);
112 custom_p2p_8023_primary_dest_valid = true;
113 }
114 if (memcmp(init->pdelay_p2p_8023_dest, zero_mac, mac_size)) {
115 memcpy(custom_p2p_8023_pdel_dest, &init->pdelay_p2p_8023_dest, mac_size);
116 custom_p2p_8023_pdel_dest_valid = true;
117 }
118
119 // store configuration
120 TP = init->tp;
121 DM = init->dm;
122
123 // join new IGMP group
124 ptp_nsd_igmp_join_leave(true);
125
126 // unlock LWIP core
127 UNLOCK_TCPIP_CORE();
128}

142void ptp_nsd_transmit_msg(RawPtpMessage *pMsg, uint32_t uid) {
143 if (pMsg == NULL) {
144 MSG("NULL!!!\n");
145 return;
146 }
147
148 PtpMessageClass mc = pMsg->tx_mc;
149 PtpMessageType mt = pMsg->tx_mt;
150
151 // allocate buffer
152 struct pbuf *p = NULL;
153 p = pbuf_alloc((TP == PTP_TP_IPv4) ? PBUF_TRANSPORT : PBUF_LINK, pMsg->size, PBUF_RAM);
154
155 /* pbuf_alloc() returns NULL when lwIP's heap is exhausted, which sustained receive pressure
156 * will do. Dereferencing it here faulted the board rather than dropping a message -- and a
157 * transmit that cannot get memory is exactly the case where dropping is right. */
158 if (p == NULL) {
159 return;
160 }
161
162 // fill buffer
163 memcpy(p->payload, pMsg->data, pMsg->size);
164
165 // set transmit callback
166 p->tag = (void *)uid;
167 p->tx_cb = ptp_transmit_cb;
168
169 // lock LWIP core
170 LOCK_TCPIP_CORE();
171
172 // is it a Peer Delay Mechanism related message?
173 bool isPDel_ = (mt == PTP_MT_PDelay_Req) || (mt == PTP_MT_PDelay_Resp) || (mt == PTP_MT_PDelay_Resp_Follow_Up);
174
175 // narrow down by transport type
176 if (TP == PTP_TP_IPv4) {
177 struct udp_pcb *conn = (mc == PTP_MC_EVENT) ? PTP_L4_PRIMARY_EVENT : PTP_L4_PRIMARY_GENERAL; // select connection by message type
178 uint16_t port = (mc == PTP_MC_EVENT) ? PTP_PORT_EVENT : PTP_PORT_GENERAL; // select port by message class
179 ip_addr_t ipaddr = isPDel_ ? PTP_IGMP_PEER_DELAY : PTP_IGMP_PRIMARY; // select destination IP-address by PDel*/primary message types
180 udp_sendto(conn, p, &ipaddr, port); // send packet
181 } else if (TP == PTP_TP_802_3) {
182 const uint8_t *ethaddr = isPDel_ ?
183 (custom_p2p_8023_pdel_dest_valid ? custom_p2p_8023_pdel_dest : PTP_ETHERNET_PEER_DELAY) :
184 (custom_p2p_8023_primary_dest_valid ? custom_p2p_8023_primary_dest : PTP_ETHERNET_PRIMARY); // select destination address by PDel*/primary message types
185 ethernet_output(netif_default, p, (struct eth_addr *)netif_default->hwaddr, (struct eth_addr *)ethaddr, ETHERTYPE_PTP);
186 }
187
188 // unlock LWIP core
189 UNLOCK_TCPIP_CORE();
190
191 pbuf_free(p); // release buffer
192}
